      document.getElementById('sdVisitors').textContent = sdVisitorsData.length;
      document.getElementById('sdSubs').textContent = sdUsersData.filter(function(u) { return u.subscribed; }).length;
      document.getElementById('sdToday').textContent = sdUsersData.filter(function(u) {
        return u.createdAt && u.createdAt.seconds && isToday(new Date(u.createdAt.seconds * 1000).toISOString());
      }).length;

      // البلدان
      var countries = {};
      sdUsersData.forEach(function(u) {
        if (u.country) countries[u.country] = (countries[u.country] || 0) + 1;
      });
      var sorted = Object.keys(countries).map(function(k) { return { name: k, count: countries[k] }; }).sort(function(a, b) { return b.count - a.count; }).slice(0, 5);
      var maxC = sorted[0] ? sorted[0].count : 1;
      var medals = ['🥇', '🥈', '🥉', '4️⃣', '5️⃣'];
      document.getElementById('sdCountries').innerHTML = sorted.length ? sorted.map(function(c, i) {
        return '<div style="margin-bottom:10px;"><div style="display:flex;justify-content:space-between;margin-bottom:4px;"><span style="color:#fbbf24;font-weight:900;font-size:13px;">' + medals[i] + ' ' + c.name + '</span><span style="color:#4ade80;font-weight:900;font-size:13px;">' + c.count + ' 👤</span></div><div style="background:rgba(51,65,85,0.4);height:8px;border-radius:10px;overflow:hidden;"><div style="height:100%;width:' + Math.max(c.count / maxC * 100, 8) + '%;background:linear-gradient(90deg,#fbbf24,#f59e0b);border-radius:10px;"></div></div></div>';
      }).join('') : '<p style="color:#94a3b8;">لا توجد بيانات</p>';

      sdRenderUsers();
    } catch(e) {
      console.error(e);
      document.getElementById('sdUsersList').innerHTML = '<p style="color:#fca5a5;">❌ خطأ: ' + e.message + '</p>';
    }
  };

  window.sdRenderUsers = function() {
    var search = (document.getElementById('sdSearch').value || '').toLowerCase().trim();
    var filtered = sdUsersData.filter(function(u) {
      if (sdFilter === 'sub' && !u.subscribed) return false;
      if (sdFilter === 'free' && u.subscribed) return false;
      if (search) {
        var hay = ((u.name || '') + ' ' + (u.email || '') + ' ' + (u.whatsapp || '') + ' ' + (u.profession || '')).toLowerCase();
        if (hay.indexOf(search) === -1) return false;
      }
      return true;
    });

    document.getElementById('sdCountUsers').textContent = filtered.length;
    if (!filtered.length) {
      document.getElementById('sdUsersList').innerHTML = '<p style="color:#94a3b8;text-align:center;padding:20px;">لا نتائج</p>';
      return;
    }

    var toShow = sdShowAll ? filtered : filtered.slice(0, 5);
    var html = toShow.map(function(u, i) {
      return '<div class="sd-user">' +
        '<div class="u-name">' + (i + 1) + '. ' + (u.name || '-') + (u.subscribed ? ' 💎' : '') + '</div>' +
        '<div class="u-info">📧 ' + (u.email || '-') + ' • 📱 ' + (u.whatsapp || '-') + '</div>' +
        '<div class="u-info">💼 ' + (u.profession || '-') + ' • 🎯 ' + (u.specialty || '-') + '</div>' +
        '<div class="u-info">🌍 ' + (u.country || '-') + ' • 🏙️ ' + (u.city || '-') + ' • 🏷️ ' + (u.category || '-') + '</div>' +
        '<div style="margin-top:8px;"><a href="https://wa.me/' + (u.whatsapp || '').replace(/[^0-9]/g, '') + '" target="_blank" class="sd-btn" style="background:#22c55e;text-decoration:none;display:inline-block;padding:6px 14px;font-size:12px;">💬 واتساب</a></div>' +
      '</div>';
    }).join('');

    if (filtered.length > 5 && !sdShowAll) {
      html += '<button onclick="sdShowAll=true;sdRenderUsers()" class="sd-btn" style="background:#3b82f6;width:100%;padding:12px;margin-top:10px;">👁️ عرض كل المسجلين (' + filtered.length + ')</button>';
    }
    if (sdShowAll && filtered.length > 5) {
      html += '<button onclick="sdShowAll=false;sdRenderUsers()" class="sd-btn" style="background:#f59e0b;width:100%;padding:12px;margin-top:10px;">🔼 عرض 5 فقط</button>';
    }

    document.getElementById('sdUsersList').innerHTML = html;
  };

  window.sdExportUsers = function() {
    if (!sdUsersData.length) return alert('⚠️ لا بيانات');
    var csv = 'الاسم,المهنة,الاختصاص,البريد,الواتساب,البلد,المدينة,الفئة,التاريخ\n';
    sdUsersData.forEach(function(u) {
      csv += '"' + (u.name || '') + '","' + (u.profession || '') + '","' + (u.specialty || '') + '","' + (u.email || '') + '","' + (u.whatsapp || '') + '","' + (u.country || '') + '","' + (u.city || '') + '","' + (u.category || '') + '","' + (u.date || '') + '"\n';
    });
    var blob = new Blob(['\ufeff' + csv], { type: 'text/csv;charset=utf-8;' });
    var a = document.createElement('a');
    a.href = URL.createObjectURL(blob);
    a.download = 'المسجلون_' + new Date().toISOString().slice(0, 10) + '.csv';
    a.click();
  };
